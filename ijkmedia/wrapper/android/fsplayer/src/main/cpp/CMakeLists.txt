cmake_minimum_required(VERSION 3.22)

project(fsplayer)

set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wno-int-conversion")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wno-int-conversion")

# ABI -> 库子目录后缀
if (ANDROID_ABI STREQUAL "arm64-v8a")
    set(ARCH "arm64")
elseif (ANDROID_ABI STREQUAL "armeabi-v7a")
    set(ARCH "armv7a")
elseif (ANDROID_ABI STREQUAL "x86")
    set(ARCH "x86")
elseif (ANDROID_ABI STREQUAL "x86_64")
    set(ARCH "x86_64")
else()
    message(FATAL_ERROR "Unsupported ANDROID_ABI: ${ANDROID_ABI}")
endif()

# 从 wrapper/android/fsplayer/src/main/cpp 定位 ijkmedia 和仓库根目录。
get_filename_component(ijkmedia "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
get_filename_component(REPO_ROOT "${ijkmedia}/.." ABSOLUTE)

# ijkversion.h 是 version.sh 为 Apple 构建生成的产物，而且被 .gitignore 忽略，
# 所以全新 checkout 里并不存在 —— ijkplayer.c / ff_ffplay.c 都 include 它，
# CI 上会直接 fatal error: 'ijkversion.h' file not found。
# 这里按同样的格式为 Android 生成一份，并放在 include 路径最前面（优先于本地残留）。
file(STRINGS ${REPO_ROOT}/version.sh _fs_ver_line REGEX "^VERSION_NAME=")
string(REGEX REPLACE "^VERSION_NAME=([^ \t\r\n]+).*" "\\1" FS_VERSION "${_fs_ver_line}")
if(NOT FS_VERSION)
    set(FS_VERSION "unknown")
endif()
set(FS_VER_DIR ${CMAKE_CURRENT_BINARY_DIR}/fsgenerated)
file(WRITE ${FS_VER_DIR}/ijkversion.h
"/* Automatically generated by the android build, do not manually edit! */
#ifndef IJKVERSION_H
#define IJKVERSION_H
#define FSPLAYER_VERSION \"${FS_VERSION}-android\"
#endif /* IJKVERSION_H */
")
message(STATUS "generated ijkversion.h with FSPLAYER_VERSION=\"${FS_VERSION}-android\"")

# 三方库（FFToolChain 预编译产物）
# FFmpeg 有两种落地路径：
#   1. CI 预编译包解压后（main.sh install -p android -l ffmpeg8）：universal/ffmpeg/ffmpeg-${ARCH}
#   2. 本地 main.sh compile 直接编译：product/android/ffmpeg-${ARCH}
# 两者都认，优先用 CI 预编译产物。
set(UNI_DIR ${REPO_ROOT}/FFToolChain/build/product/android/universal)
set(FFMPEG_DIR "")
foreach(cand
        ${REPO_ROOT}/FFToolChain/build/product/android/universal/ffmpeg/ffmpeg-${ARCH}
        ${REPO_ROOT}/FFToolChain/build/product/android/ffmpeg-${ARCH})
    if(EXISTS ${cand}/include/libffmpeg/config.h)
        set(FFMPEG_DIR ${cand})
    endif()
endforeach()
if(NOT FFMPEG_DIR)
    message(FATAL_ERROR "FFmpeg 产物缺失，请先运行：./FFToolChain/main.sh install -p android -l ffmpeg8")
endif()
message(STATUS "FFmpeg product: ${FFMPEG_DIR}")

foreach(cand "${UNI_DIR}/yuv/yuv-${ARCH}"
             "${REPO_ROOT}/FFToolChain/build/product/android/yuv-${ARCH}")
    if(EXISTS "${cand}/include/libyuv/convert_argb.h" AND EXISTS "${cand}/lib/libyuv.a")
        set(YUV_DIR "${cand}")
        break()
    endif()
endforeach()
if(NOT YUV_DIR)
    message(FATAL_ERROR "libyuv 产物缺失，请先运行：./FFToolChain/main.sh install -p android -l yuv")
endif()

file(GLOB_RECURSE SOURCE_IJKJ4A ${ijkmedia}/ijkj4a/*.c ${ijkmedia}/ijkj4a/*.cpp)
file(GLOB_RECURSE SOURCE_SDL CONFIGURE_DEPENDS ${ijkmedia}/ijksdl/*.c ${ijkmedia}/ijksdl/*.cpp)
file(GLOB_RECURSE SOURCE_IJKPLAYER ${ijkmedia}/ijkplayer/*.c ${ijkmedia}/ijkplayer/*.cpp)
file(GLOB_RECURSE SOURCE_SOUNDTOUCH ${ijkmedia}/ijksoundtouch/*.c)

# 排除苹果平台专属（.c 的 heic 等）
list(FILTER SOURCE_SDL EXCLUDE REGEX "/apple/")
list(FILTER SOURCE_IJKPLAYER EXCLUDE REGEX "/apple/")
# 排除依赖 ffmpeg 内部头 http.h（安卓预编译 ffmpeg 未导出）的 HLS/segment
list(FILTER SOURCE_IJKPLAYER EXCLUDE REGEX "ijkhls")
list(FILTER SOURCE_IJKPLAYER EXCLUDE REGEX "ijksegment")

add_library(fsplayer SHARED ${SOURCE_IJKJ4A} ${SOURCE_SDL} ${SOURCE_IJKPLAYER} ${SOURCE_SOUNDTOUCH})
target_include_directories(fsplayer PRIVATE
    "${FS_VER_DIR}"
    "${ijkmedia}"
    "${ijkmedia}/ijkj4a"
    "${ijkmedia}/ijksdl"
    "${ijkmedia}/ijksdl/ffmpeg"
    "${ijkmedia}/ijkplayer"
    "${FFMPEG_DIR}/include"
    "${UNI_DIR}/ass/ass-${ARCH}/include"
    "${YUV_DIR}/include")

# 排查用（二分定位）：./gradlew :fsplayer:assembleDebug -PfsNoVulkan=1
# 让 SDL_VoutAndroid_CreateForVulkan 直接返回 NULL，用来判断崩溃是否来自 Vulkan 渲染器本身。
if (fsNoVulkan)
    add_compile_definitions(FS_NO_VULKAN=1)
endif()

set(FFMPEG_LIB ${FFMPEG_DIR}/lib)
add_library(ijkffmpeg-lib INTERFACE)
target_link_libraries(ijkffmpeg-lib INTERFACE
    ${FFMPEG_LIB}/libavdevice.a
    ${FFMPEG_LIB}/libavformat.a
    ${FFMPEG_LIB}/libavcodec.a
    ${FFMPEG_LIB}/libavfilter.a
    ${FFMPEG_LIB}/libswresample.a
    ${FFMPEG_LIB}/libswscale.a
    ${FFMPEG_LIB}/libavutil.a)

set(ASS_LIB       ${UNI_DIR}/ass/ass-${ARCH}/lib/libass.a)
set(FT_LIB        ${UNI_DIR}/freetype/freetype-${ARCH}/lib/libfreetype.a)
set(FRIBIDI_LIB   ${UNI_DIR}/fribidi/fribidi-${ARCH}/lib/libfribidi.a)
set(HARFBUZZ_LIB  ${UNI_DIR}/harfbuzz/harfbuzz-${ARCH}/lib/libharfbuzz.a)
set(UNIBREAK_LIB  ${UNI_DIR}/unibreak/unibreak-${ARCH}/lib/libunibreak.a)
set(OPENSSL_SSL   ${UNI_DIR}/openssl/openssl-${ARCH}/lib/libssl.a)
set(OPENSSL_CRYPTO ${UNI_DIR}/openssl/openssl-${ARCH}/lib/libcrypto.a)

# ffmpeg 编译时启用并链接的三方库（pkgconfig Libs 里列出的）
set(DAV1D_LIB     ${UNI_DIR}/dav1d/dav1d-${ARCH}/lib/libdav1d.a)
set(OPUS_LIB      ${UNI_DIR}/opus/opus-${ARCH}/lib/libopus.a)
set(UAVS3D_LIB    ${UNI_DIR}/uavs3d/uavs3d-${ARCH}/lib/libuavs3d.a)
set(XML2_LIB      ${UNI_DIR}/xml2/xml2-${ARCH}/lib/libxml2.a)
set(BLURAY_LIB    ${UNI_DIR}/bluray/bluray-${ARCH}/lib/libbluray.a)
# 不链 libdvdnav/libdvdread：它们是 GPL，会把 FFmpeg 从 LGPL 拉成 nonfree；
# 预编译 product 里也不再有它们（ffconfig 的 auto-detect 只在探测到才启用）
set(FONTCONFIG_LIB ${UNI_DIR}/fontconfig/fontconfig-${ARCH}/lib/libfontconfig.a)

# FFToolChain 将 SMB2 放在与 FFmpeg 同级的目录。
find_file(SMB2_LIB NAMES libsmb2.a
    PATHS "${UNI_DIR}/smb2/smb2-${ARCH}/lib" "${UNI_DIR}/../smb2-${ARCH}/lib"
    NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH NO_CACHE)
if(NOT SMB2_LIB)
    message(FATAL_ERROR "libSMB2 静态库缺失，请使用包含 libsmb2.a 的 FFToolChain Android FFmpeg 预编译包")
endif()

find_library(z-lib z REQUIRED)
find_library(log-lib log)
find_library(android-lib android)
find_library(egl-lib EGL REQUIRED)
find_library(gles-lib GLESv3 REQUIRED)
find_library(jnigraphics-lib jnigraphics)
find_library(m-lib m)
find_library(mediandk-lib mediandk)
find_library(dl-lib dl)

target_link_libraries(fsplayer
    "-Wl,--start-group"
    ijkffmpeg-lib
    "${YUV_DIR}/lib/libyuv.a"
    ${DAV1D_LIB}
    ${OPUS_LIB}
    ${UAVS3D_LIB}
    ${XML2_LIB}
    ${BLURAY_LIB}
    ${OPENSSL_SSL}
    ${OPENSSL_CRYPTO}
    ${ASS_LIB}
    ${FT_LIB}
    ${FRIBIDI_LIB}
    ${HARFBUZZ_LIB}
    ${UNIBREAK_LIB}
    ${FONTCONFIG_LIB}
    "${SMB2_LIB}"
    "-Wl,--end-group"
    ${z-lib}
    ${dl-lib}
    ${mediandk-lib}
    ${log-lib}
    ${android-lib}
    ${egl-lib}
    ${gles-lib}
    ${jnigraphics-lib}
    ${m-lib}
    "-Wl,-z,max-page-size=16384"
    "-Wl,-z,common-page-size=16384"
    "-Wl,-Bsymbolic")
